cmake_minimum_required(VERSION 3.22)
project(dota_item_merge VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(BUILD_GAME "Build the Raylib desktop game" ON)
include(CTest)
include(cmake/Box2D.cmake)
add_library(merge_physics STATIC src/game.c)
target_link_libraries(merge_physics PUBLIC box2d)
target_include_directories(merge_physics PUBLIC src)
if(NOT MSVC)
    target_compile_options(merge_physics PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(merge_physics PUBLIC m)
endif()
if(BUILD_GAME)
    find_package(raylib 5.5 QUIET)
    if(NOT raylib_FOUND)
        include(FetchContent)
        set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz)
        FetchContent_MakeAvailable(raylib)
    endif()
    add_executable(dota-item-merge src/main.c)
    target_link_libraries(dota-item-merge PRIVATE merge_physics raylib)
    add_custom_target(game-assets ALL
        COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:dota-item-merge>/assets"
        COMMENT "Copying bundled item artwork" VERBATIM)
    add_dependencies(dota-item-merge game-assets)
    if(NOT MSVC)
        target_compile_options(dota-item-merge PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
if(BUILD_TESTING)
    add_executable(physics-tests tests/test_game.c)
    target_link_libraries(physics-tests PRIVATE merge_physics)
    if(NOT MSVC)
        target_compile_options(physics-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME physics COMMAND physics-tests)
endif()
